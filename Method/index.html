Assignment 3: Shopping Cart & Membership Discount

Scenario: An online shopping application calculates the final bill based on the purchase amount and membership type.

Concepts: Multi-product summation method (calculateTotal), tiered discount application (calculateDiscount: Gold 20%, Silver 10%, Regular 5%), and final bill calculation (calculateFinalAmount).

package com.java.assignment;

public class ShoppingCart {

    double item1;
    double item2;
    double item3;
    String membership;

    double calculateTotal(double p1, double p2, double p3) {
        double total = p1 + p2 + p3;
        return total;
    }

    double calculateDiscount(double total, String membership) {
        double discount = 0;
        if (membership.equalsIgnoreCase("Gold")) {
            discount = total * 0.20;
        } else if (membership.equalsIgnoreCase("Silver")) {
            discount = total * 0.10;
        } else if (membership.equalsIgnoreCase("Regular")) {
            discount = total * 0.05;
        }
        return discount;
    }

    double calculateFinalAmount(double total, double discount) {
        double finalAmount = total - discount;
        return finalAmount;
    }

    void displayDetails(double total, double discount, double finalAmount) {
        System.out.println("Total Amount: " + total);
        System.out.println("Discount: " + discount);
        System.out.println("Final Amount: " + finalAmount);
    }

    public static void main(String[] args) {
        ShoppingCart cart = new ShoppingCart();

        cart.item1 = 2000;
        cart.item2 = 1500;
        cart.item3 = 500;
        cart.membership = "Gold";

        double total = cart.calculateTotal(cart.item1, cart.item2, cart.item3);
        double discount = cart.calculateDiscount(total, cart.membership);
        double finalAmount = cart.calculateFinalAmount(total, discount);

        cart.displayDetails(total, discount, finalAmount);
    }
}

Assignment 4: Progressive Electricity Bill Calculation

Scenario: An electricity board calculates a household bill based on units consumed using progressive slabs.

Concepts: Progressive tiered calculation method (calculateBill: 1st 100 @ ₹2, next 100 @ ₹3, remainder @ ₹5), fixed surcharge method (addFixedCharge: +₹50), and summary display.

package com.java.assignment;

public class ElectricityBill {

    int units;

    double calculateBill(int units) {
        double bill = 0;
        if (units <= 100) {
            bill = units * 2;
        } else if (units <= 200) {
            bill = (100 * 2) + ((units - 100) * 3);
        } else {
            bill = (100 * 2) + (100 * 3) + ((units - 200) * 5);
        }
        return bill;
    }

    double addFixedCharge(double billAmount) {
        double finalBill = billAmount + 50;
        return finalBill;
    }

    void displayDetails(double energyCharge, double fixedCharge, double finalBill) {
        System.out.println("Energy Charge: " + energyCharge);
        System.out.println("Fixed Charge: " + fixedCharge);
        System.out.println("Final Bill: " + finalBill);
    }

    public static void main(String[] args) {
        ElectricityBill eb = new ElectricityBill();

        eb.units = 250;

        double energyCharge = eb.calculateBill(eb.units);
        double finalBill = eb.addFixedCharge(energyCharge);
        double fixedCharge = 50;

        eb.displayDetails(energyCharge, fixedCharge, finalBill);
    }
}

Assignment 5: Banking Operations (Deposit, Withdraw, Balance)

Scenario: A bank needs a simple application to perform deposits, withdrawals, and balance checks without using constructors.

Concepts: Stateful transaction methods (deposit, withdraw with validation), status check method (checkBalance), and summary display (displayTransactionSummary).

package com.java.assignment;

public class BankOperations {

    double InitialBalance;
    double DepositAmount;
    double WithdrawalAmount;

    double deposit(double Balance, double DepositAmount) {
        double UpdatedBalance = Balance + DepositAmount;
        return UpdatedBalance;
    }

    double withdraw(double Balance, double WithdrawalAmount) {
        if (Balance >= WithdrawalAmount) {
            Balance = Balance - WithdrawalAmount;
        } 
        else {
            System.out.println("Insufficient Balance");
        }

        return Balance;
    }

    void checkBalance(double Balance) {
        System.out.println("Current Balance: " + Balance);
    }

    void displayTransactionSummary(double DepositAmount, double WithdrawalAmount, double FinalBalance) {
        System.out.println("Deposited: " + DepositAmount);
        System.out.println("Withdrawn: " + WithdrawalAmount);
        System.out.println("Final Balance: " + FinalBalance);
    }

    public static void main(String[] args) {
        BankOperations Bank = new BankOperations();

        Bank.InitialBalance = 5000;
        Bank.DepositAmount = 2000;
        Bank.WithdrawalAmount = 1500;

        System.out.println("Initial Balance: " + Bank.InitialBalance);

        double Balance = Bank.deposit(
            Bank.InitialBalance, Bank.DepositAmount
        );

        Balance = Bank.withdraw(Balance, Bank.WithdrawalAmount);

        Bank.checkBalance(Balance);

        Bank.displayTransactionSummary(
            Bank.DepositAmount, Bank.WithdrawalAmount, Balance
        );
    }
}
