Assignment 1: ATM PIN Verification & Withdrawal

Scenario: Simulate an ATM withdrawal where PIN verification is validated first, followed by sufficient funds check.

Control Flow Concept: Nested if-else statement — the outer if validates the PIN code, while the inner if checks whether current balance is sufficient for withdrawal.

package com.java.Assignment2;

public class ATM {

    int CorrectPIN;
    int EnteredPIN;
    double Balance;
    double WithdrawalAmount;

    void withdraw(int EnteredPIN, double Balance, double WithdrawalAmount) {
        if (EnteredPIN == CorrectPIN) {
            if (Balance >= WithdrawalAmount) {
                Balance = Balance - WithdrawalAmount;
                System.out.println("Withdrawal Successful");
                System.out.println("Remaining Balance: " + Balance);
            } else {
                System.out.println("Insufficient Balance");
            }
        } else {
            System.out.println("Incorrect PIN");
        }
    }

    public static void main(String[] args) {
        ATM atm = new ATM();

        atm.CorrectPIN = 1234;
        atm.EnteredPIN = 1234;
        atm.Balance = 5000;
        atm.WithdrawalAmount = 2000;

        atm.withdraw(atm.EnteredPIN, atm.Balance, atm.WithdrawalAmount);
    }
}

Assignment 2: Progressive Electricity Bill Calculation

Scenario: Calculate consumer power charges across graded slab units (≤100 units @ ₹2, ≤200 units @ ₹3, >200 units @ ₹5).

Control Flow Concept: Multi-way branching with if - else if - else ladder to apply progressive tariffs across slab boundaries.

package com.java.Assignment2;

public class ElectricityBill {

    double calculateBill(double units) {
        double bill;

        if (units <= 100) {
            bill = units * 2;
        } else if (units <= 200) {
            bill = (100 * 2) + (units - 100) * 3;
        } else {
            bill = (100 * 2) + (100 * 3) + (units - 200) * 5;
        }

        return bill;
    }

    public static void main(String[] args) {
        ElectricityBill eb = new ElectricityBill();

        double units = 250;
        double bill = eb.calculateBill(units);

        System.out.println("Units Consumed: " + units);
        System.out.println("Total Bill: " + bill);
    }
}

Assignment 3: User Login Verification with Limited Attempts

Scenario: Check password correctness with up to 3 attempts, terminating immediately on valid authentication.

Control Flow Concept: while loop iteration with loop counter and the break jump statement to terminate the loop upon successful login.

package com.java.Assignment2;

public class LoginAttempt {

    public static void main(String[] args) {
        String correctPassword = "java123";
        String enteredPassword = "something";

        int attempts = 1;

        while (attempts <= 3) {
            if (enteredPassword.equals(correctPassword)) {
                System.out.println("Login Successful");
                break;
            } else {
                System.out.println("Invalid Password");
            }

            attempts++;
        }
    }
}

Assignment 4: Loop 1 to 30 and Skip Multiples of 3

Scenario: Iterate through numbers 1 to 30, skipping all values divisible by 3.

Control Flow Concept: for loop iteration combined with the continue statement to bypass the remainder of the iteration when i % 3 == 0.

package com.java.Assignment2;

public class SkipMultiples {

    public static void main(String[] args) {

        for (int i = 1; i <= 30; i++) {
            if (i % 3 == 0) {
                continue;
            }

            System.out.println(i);
        }
    }
}

Assignment 5: Student Grade Evaluation with Input Validation

Scenario: Evaluate letter grade from numerical score with boundary validation (<0 or >100: Invalid, ≥90: A, ≥75: B, ≥50: C, else: Fail).

Control Flow Concept: Logical operators (||, >=) in sequential if-else checks to guarantee input sanitization and correct tier mapping.

package com.java.Assignment2;

public class StudentGrade {

    void calculateGrade(double marks) {
        if (marks < 0 || marks > 100) {
            System.out.println("Invalid Marks");
        } else if (marks >= 90) {
            System.out.println("Grade: A");
        } else if (marks >= 75) {
            System.out.println("Grade: B");
        } else if (marks >= 50) {
            System.out.println("Grade: C");
        } else {
            System.out.println("Fail");
        }
    }

    public static void main(String[] args) {
        StudentGrade sg = new StudentGrade();

        double marks = 84;
        sg.calculateGrade(marks);
    }
}
